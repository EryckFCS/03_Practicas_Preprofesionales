}

% Color formal de texto (Negro / Escala de grises formal APA)
\definecolor{APABlack}{RGB}{0, 0, 0}
\definecolor{APAGray}{RGB}{60, 60, 60}
\definecolor{APALink}{RGB}{0, 45, 98} % Azul sobrio para enlaces

\color{APABlack}

% --- Interlineado y Párrafos APA 7 ---
\onehalfspacing
\setlength{\parindent}{1.27cm} % Sangría de primera línea reglamentaria de 0.5 pulgadas
\setlength{\parskip}{0pt}       % APA 7 no deja espacio adicional entre párrafos normales

% --- Encabezados y Números de Página (APA 7) ---
\clearpairofpagestyles
\addtokomafont{pageheadfoot}{\normalfont\footnotesize\color{APAGray}}

\addtokomafont{subsection}{\raggedright\large\bfseries}
\addtokomafont{subsubsection}{\raggedright\normalsize\bfseries\itshape}

% --- Formato Reglamentario de Tablas y Figuras APA 7 ---
\DeclareCaptionLabelSeparator{newline}{\\ \vspace{2pt}}
\captionsetup[table]{
  format=plain,
  position=top,
  labelsep=newline,
  justification=RaggedRight,
  singlelinecheck=false,
  labelfont={bf},
  textfont={it},
  font={small,singlespacing},
  skip=4pt
}

\captionsetup[figure]{
  format=plain,
  position=bottom,
  labelsep=newline,
  justification=RaggedRight,
  singlelinecheck=false,
  labelfont={bf},
  textfont={it},
  font={small,singlespacing},
  skip=4pt
}

% --- Comando APA 7 para Notas de Tabla / Fuentes ---

  {\raggedright\singlespacing\footnotesize \textit{Nota.} #1\par\vspace{6pt}}%

}
