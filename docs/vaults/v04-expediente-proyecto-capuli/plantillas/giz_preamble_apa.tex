\newcommand{\Lpagenumber}{%
  \begingroup
    \dimendef\margin=0
    \ifodd\value{page}
      \margin=\oddsidemargin
    \else
      \margin=\evensidemargin
    \fi
    \raisebox{%
      \dimexpr
      -\topmargin-\headheight-\headsep-0.45\textwidth+\ajustepiehorizontal
      \relax
    }[0pt][0pt]{%
      \rlap{%
        \hspace{\dimexpr
          \margin+\textheight+\footskip+\bajarpiehorizontal
        \relax}%
        \llap{\rotatebox{90}{\thepage}}%
      }%
    }%
  \endgroup
}

\AddToHook{shipout/background}{%
  \ifTablaHorizontal
    \hbox{\Lpagenumber}%
  \fi
}

\AddToHook{shipout/before}{%

  \fi
}
